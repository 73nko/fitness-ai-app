```typescript
// src/hooks/useTraining.ts
import { useState, useEffect, useCallback } from 'react';
import { apiClient } from '../api/restClient';
import { TrainingPlan } from '@fitnessai/types';

export function useTraining() {
  const [plan, setPlan] = useState<TrainingPlan | null>(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);
  
  const fetchPlan = useCallback(async () => {
    setLoading(true);
    setError(null);
    
    try {
      const trainingPlan = await apiClient.getTrainingPlan();
      setPlan(trainingPlan);
      return trainingPlan;
    } catch (err) {
      setError(err.message || 'Failed to fetch training plan');
      throw err;
    } finally {
      setLoading(false);
    }
  }, []);
  
  useEffect(() => {
    fetchPlan().catch(err => console.error('Initial plan fetch failed', err));
  }, [fetchPlan]);
  
  const updatePlan = useCallback(async (updates) => {
    setLoading(true);
    setError(null);
    
    try {
      const updatedPlan = await apiClient.updateTrainingPlan(updates);
      setPlan(updatedPlan);
      return updatedPlan;
    } catch (err) {
      setError(err.message || 'Failed to update training plan');
      throw err;
    } finally {
      setLoading(false);
    }
  }, []);
  
  const submitFeedback = useCallback(async (sessionId, feedback) => {
    try {
      return await apiClient.submitTrainingFeedback(sessionId, feedback);
    } catch (err) {
      setError(err.message || 'Failed to submit feedback');
      throw err;
    }
  }, []);
  
  const getSuggestions = useCallback(async () => {
    try {
      return await apiClient.getTrainingSuggestions();
    } catch (err) {
      setError(err.message || 'Failed to get suggestions');
      throw err;
    }
  }, []);
  
  return { 
    plan, 
    loading, 
    error, 
    fetchPlan, 
    updatePlan, 
    submitFeedback, 
    getSuggestions 
  };
}
```
