Example implementation:
```typescript
// src/api/rest/routes/training.ts
import { FastifyInstance } from 'fastify';
import { TrainingService } from '../../../services/training';
import { UpdateTrainingPlanSchema } from '@fitnessai/types';
import { authenticate } from '../middleware/auth';

export async function trainingRoutes(fastify: FastifyInstance) {
  const trainingService = new TrainingService();
  
  fastify.get('/plan', {
    preHandler: [authenticate],
    handler: async (request, reply) => {
      const userId = request.user.id;
      const plan = await trainingService.getCurrentPlan(userId);
      if (!plan) {
        return reply.status(404).send({
          error: 'NotFound',
          message: 'Training plan not found'
        });
      }
      return reply.send(plan);
    }
  });
  
  fastify.post('/plan/update', {
    schema: {
      body: UpdateTrainingPlanSchema
    },
    preHandler: [authenticate],
    handler: async (request, reply) => {
      const userId = request.user.id;
      const updatedPlan = await trainingService.updatePlan(userId, request.body);
      return reply.send(updatedPlan);
    }
  });
}
```
