```typescript
// tests/integration/rest-api.test.ts
import { build } from '../../src/app';
import { test, beforeAll, afterAll, expect, describe } from 'vitest';
import { createTestUser, generateAuthToken } from '../helpers/auth';

describe('REST API Integration Tests', () => {
  let app;
  let testUser;
  let authToken;
  
  beforeAll(async () => {
    app = build();
    await app.ready();
    
    // Create test user and get auth token
    testUser = await createTestUser();
    authToken = generateAuthToken(testUser);
  });
  
  afterAll(async () => {
    await app.close();
  });
  
  describe('Authentication Endpoints', () => {
    test('POST /api/auth/login - should return token for valid credentials', async () => {
      const response = await app.inject({
        method: 'POST',
        url: '/api/auth/login',
        payload: {
          email: testUser.email,
          password: 'password123' // Test password
        }
      });
      
      expect(response.statusCode).toBe(200);
      const body = JSON.parse(response.body);
      expect(body).toHaveProperty('token');
      expect(body).toHaveProperty('user');
      expect(body.user.id).toBe(testUser.id);
    });
    
    // More auth tests...
  });
  
  describe('Training Plan Endpoints', () => {
    test('GET /api/training/plan - should return user training plan', async () => {
      const response = await app.inject({
        method: 'GET',
        url: '/api/training/plan',
        headers: {
          Authorization: `Bearer ${authToken}`
        }
      });
      
      expect(response.statusCode).toBe(200);
      const body = JSON.parse(response.body);
      expect(body).toHaveProperty('id');
      expect(body).toHaveProperty('weeks');
      // Verify plan structure matches expected format
    });
    
    // More training endpoints tests...
  });
  
  // Additional test suites for other endpoint groups
});
```
