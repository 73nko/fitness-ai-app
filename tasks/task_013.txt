2. Ensure both APIs access the same service layer
3. Implement logging to track usage of each API

Example implementation:
```typescript
// src/server.ts
import fastify from 'fastify';
import { setupGrpcServer } from './grpc';
import { setupRestApi } from './api/rest';
import { logger } from './utils/logger';

async function startServer() {
  const app = fastify({
    logger: true
  });
  
  // Set up REST API
  setupRestApi(app);
  
  // Set up gRPC server (existing code)
  const grpcServer = setupGrpcServer();
  
  // Start both servers
  try {
    const port = process.env.PORT || 3000;
    await app.listen({ port });
    logger.info(`REST API listening on port ${port}`);
    
    // gRPC server is already running from setupGrpcServer
    
    // Add shutdown handler
    const shutdown = async () => {
      logger.info('Shutting down servers...');
      await app.close();
      grpcServer.forceShutdown();
      process.exit(0);
    };
    
    process.on('SIGTERM', shutdown);
    process.on('SIGINT', shutdown);
  } catch (err) {
    logger.error('Error starting servers:', err);
    process.exit(1);
  }
}

startServer();
```
