// src/config/featureFlags.ts
import AsyncStorage from '@react-native-async-storage/async-storage';

export enum ApiType {
  GRPC = 'grpc',
  REST = 'rest'
}

class FeatureFlags {
  private flags: Record<string, any> = {
    apiType: ApiType.GRPC // Default to gRPC initially
  };
  
  async initialize() {
    try {
      // Load saved flags
      const savedFlags = await AsyncStorage.getItem('feature_flags');
      if (savedFlags) {
        this.flags = { ...this.flags, ...JSON.parse(savedFlags) };
      }
      
      // Could also fetch remote flags here
      // const remoteFlags = await fetchRemoteFlags();
      // this.flags = { ...this.flags, ...remoteFlags };
      
      // Save merged flags
      await this.saveFlags();
    } catch (error) {
      console.error('Failed to initialize feature flags', error);
    }
  }
  
  async saveFlags() {
    await AsyncStorage.setItem('feature_flags', JSON.stringify(this.flags));
  }
  
  getApiType(): ApiType {
    return this.flags.apiType;
  }
  
  async setApiType(type: ApiType) {
    this.flags.apiType = type;
    await this.saveFlags();
  }
}

export const featureFlags = new FeatureFlags();

// src/api/clientFactory.ts
import { featureFlags, ApiType } from '../config/featureFlags';
import { grpcClient } from './grpcClient';
import { apiClient } from './restClient';

export function getApiClient() {
  const apiType = featureFlags.getApiType();
  
  switch (apiType) {
    case ApiType.REST:
      return apiClient;
    case ApiType.GRPC:
    default:
      return grpcClient;
  }
}

// Usage in hooks
import { getApiClient } from '../api/clientFactory';

export function useAuth() {
  const client = getApiClient();
  // Use client for API calls
}
