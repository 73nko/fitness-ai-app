Example implementation:
```typescript
// src/api/rest/routes/training.ts (continued)

// Inside trainingRoutes function
fastify.get('/suggestions', {
  preHandler: [authenticate],
  handler: async (request, reply) => {
    const userId = request.user.id;
    
    try {
      const suggestions = await trainingService.generateSuggestions(userId);
      return reply.send(suggestions);
    } catch (error) {
      if (error.message === 'No training data available') {
        return reply.status(400).send({
          error: 'InsufficientData',
          message: 'Not enough training data to generate suggestions'
        });
      }
      throw error;
    }
  }
});
```
